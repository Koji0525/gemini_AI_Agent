import cProfile
import pstats
from io import StringIO

# プロファイリングの実施
def profile_function():
    pr = cProfile.Profile()
    pr.enable()
    
    # 対象の関数実行
    target_function()
    
    pr.disable()
    s = StringIO()
    ps = pstats.Stats(pr, stream=s).sort_stats('cumulative')
    ps.print_stats()
    print(s.getvalue())
