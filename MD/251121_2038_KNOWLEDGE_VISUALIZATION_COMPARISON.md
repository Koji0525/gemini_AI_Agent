#### 分析例
```python
import pandas as pd
import matplotlib.pyplot as plt
import sqlite3

# データ読み込み
conn = sqlite3.connect('knowledge_system/database/knowledge.db')
df = pd.read_sql("SELECT * FROM knowledge_entries", conn)

# 日別追加数
df['date'] = pd.to_datetime(df['created_at']).dt.date
daily = df.groupby('date').size()
daily.plot(kind='bar')
plt.title('Daily Knowledge Additions')
plt.show()

# タグ分析
tags = df['content'].str.extract(r'タグ: (.+)')[0]
tag_counts = tags.str.split(', ').explode().value_counts()
tag_counts.head(10).plot(kind='pie')
```
