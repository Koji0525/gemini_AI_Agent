from agents.self_healing.retry_manager import RetryManager, RetryConfig

class IntegratedOrchestrator:
    def __init__(self):
        # ... 既存初期化 ...
        
        # RetryManager追加
        self.retry_manager = RetryManager(
            RetryConfig(
                max_retries=3,
                base_delay=1.0,
                max_delay=60.0
            )
        )
    
    async def _execute_task_with_retry(self, task: Dict) -> Dict:
        """リトライ機能付きタスク実行"""
        async def task_func():
            return await self._execute_task(task)
        
        result = await self.retry_manager.execute_with_retry(
            task_func,
            context={'task_id': task.get('task_id')}
        )
        
        return result
```

from agents.self_healing.logging.decision_support_system import DecisionSupportSystem

class IntegratedOrchestrator:
    def __init__(self):
        # ... 既存初期化 ...
        self.decision_support = DecisionSupportSystem(self.sheets)
    
    async def _decide_fix_strategy(self, task: Dict, error: Exception):
        """ナレッジベースから最適な修正策を決定"""
        proposal = await self.decision_support.propose_solution(
            error_type=type(error).__name__,
            context={'task': task}
        )
        
        if proposal.confidence > 0.7:
            # 信頼度が高い場合は自動実行
            return await self._apply_fix(proposal)
        else:
            # 信頼度が低い場合は人間に確認
            return await self._request_human_approval(proposal)
```

# scripts/human_interaction_agent_v01_github.py

from github import Github

class HumanInteractionAgent:
    """GitHub Issuesで人間と対話"""
    
    COMMANDS = {
        '@bot stop': 'stop_execution',
        '@bot resume': 'resume_execution',
        '@bot status': 'report_status',
        '@bot approve': 'approve_proposal'
    }
    
    async def monitor_issues(self, repo_name: str):
        """Issuesを監視してコマンドを処理"""
        g = Github(os.getenv('GITHUB_TOKEN'))
        repo = g.get_repo(repo_name)
        
        while True:
            issues = repo.get_issues(
                state='open',
                labels=['bot-control']
            )
            
            for issue in issues:
                for comment in issue.get_comments():
                    command = self._parse_command(comment.body)
                    if command:
                        await self._execute_command(command)
                        comment.create_reaction('rocket')
            
            await asyncio.sleep(60)
```

# scripts/progress_dashboard_v02_realtime.py

class ProgressDashboard:
    """リアルタイム進捗ダッシュボード"""
    
    def update_progress(self, task_id: str, progress: int):
        """進捗をスプレッドシートに即座に反映"""
        self.sheets.update_cell(
            'progress_dashboard',
            row=self._find_task_row(task_id),
            col='progress',
            value=progress
        )
    
    def detect_bottleneck(self) -> List[str]:
        """ボトルネックを検出"""
        tasks = self.sheets.read_range('pm_tasks!A:I')
        stuck_tasks = []
        
        for task in tasks:
            if task['status'] == 'in_progress':
                elapsed = self._calc_elapsed(task['started_at'])
                if elapsed > 30:  # 30分以上停滞
                    stuck_tasks.append(task['task_id'])
        
        return stuck_tasks
```
