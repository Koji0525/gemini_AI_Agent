    check_status
    
    # VNCチェック
    if [ $(ps aux | grep Xvnc | grep -v grep | wc -l) -eq 0 ]; then

# test/test_tasks_gemini_with_sheets.py
import asyncio
import sys
import os
sys.path.append(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from sheets_manager import SheetsManager
from browser_controller import GeminiBrowserController
from config_utils import ConfigManager

class GeminiSheetsTester:
    def __init__(self):
        self.config = ConfigManager()
        self.sheets_manager = SheetsManager()
        self.browser = None
    
    async def initialize(self):
        """システム初期化"""
        print("🎯 Geminiシート連携テスト開始")
        
        # シートマネージャー初期化
        await self.sheets_manager.initialize()
        
        # ブラウザ初期化（VNC表示対応）
        self.browser = GeminiBrowserController()
        await self.browser.initialize(headless=False, display=':1')
        
        print("✅ システム初期化完了")
    
    async def load_and_execute_tasks(self):
        """シートからタスクを読み込んで実行"""
        try:
            # シートからタスク取得
            tasks = await self.sheets_manager.get_tasks()
            print(f"📊 シートからタスクを読み込み: {len(tasks)}件")
            
            results = []
            for task in tasks:
                result = await self.execute_single_task(task)
                results.append(result)
                
                # 結果をシートに記録
                await self.sheets_manager.update_task_result(
                    task['id'], 
                    result['status'], 
                    result.get('message', '')
                )
            
            return results
            
        except Exception as e:
            print(f"❌ タスク実行エラー: {e}")
            return []
    
    async def execute_single_task(self, task):
        """個別タスク実行"""
        try:
            task_type = task.get('type', 'unknown')
            print(f"🎯 タスク実行: {task.get('title', 'No Title')} ({task_type})")
            
            if task_type == 'gemini_chat':
                return await self.execute_gemini_chat(task)
            elif task_type == 'wordpress_post':
                return await self.execute_wordpress_task(task)
            elif task_type == 'content_generation':
                return await self.execute_content_generation(task)
            else:
                return {'status': 'skipped', 'message': f'未対応のタスクタイプ: {task_type}'}
                
        except Exception as e:
            return {'status': 'error', 'message': str(e)}
    
    async def execute_gemini_chat(self, task):
        """Geminiチャットタスク実行"""
        try:
            # ブラウザでGeminiにアクセス
            await self.browser.navigate_to_gemini()
            
            # 質問を入力
            question = task.get('content', '')
            response = await self.browser.ask_gemini(question)
            
            return {
                'status': 'completed',
                'message': f'質問: {question[:50]}... | 回答取得成功',
                'response': response
            }
        except Exception as e:
            return {'status': 'error', 'message': f'Geminiチャットエラー: {str(e)}'}
    
    async def execute_wordpress_task(self, task):
        """WordPress関連タスク実行"""
        try:
            # WordPressエージェントと連携
            from wordpress.wp_agent import WordPressAgent
            
            wp_agent = WordPressAgent()
            await wp_agent.initialize()
            
            # タスクに応じた処理
            if task.get('action') == 'create_post':
                result = await wp_agent.create_post(task)
            elif task.get('action') == 'update_settings':
                result = await wp_agent.update_settings(task)
            else:
                result = {'status': 'unknown_action'}
            
            return {
                'status': 'completed' if result.get('status') == 'success' else 'failed',
                'message': f"WordPress処理: {task.get('action')}"
            }
            
        except Exception as e:
            return {'status': 'error', 'message': f'WordPressタスクエラー: {str(e)}'}
    
    async def execute_content_generation(self, task):
        """コンテンツ生成タスク実行"""
        try:
            # コンテンツライターエージェントと連携
            from task_executor.task_executor_content import ContentTaskExecutor
            
            content_executor = ContentTaskExecutor()
            result = await content_executor.execute_content_generation(task)
            
            return {
                'status': 'completed',
                'message': f"コンテンツ生成: {task.get('topic', 'Unknown')}",
                'content': result
            }
            
        except Exception as e:
            return {'status': 'error', 'message': f'コンテンツ生成エラー: {str(e)}'}
    
    async def cleanup(self):
        """クリーンアップ"""
        if self.browser:
            await self.browser.cleanup()
        print("✅ クリーンアップ完了")

async def main():
    tester = GeminiSheetsTester()
    
    try:
        # 初期化
        await tester.initialize()
        
        # タスク実行
        results = await tester.load_and_execute_tasks()
        
        # 結果表示
        print("\n📋 実行結果サマリー:")
        for i, result in enumerate(results, 1):
            status_icon = "✅" if result['status'] == 'completed' else "❌"
            print(f"{status_icon} タスク {i}: {result['status']} - {result.get('message', '')}")
        
        print(f"\n🎉 全タスク完了: {len([r for r in results if r['status'] == 'completed'])}/{len(results)} 成功")
        
    except Exception as e:
        print(f"❌ メイン実行エラー: {e}")
    
    finally:
        await tester.cleanup()

if __name__ == "__main__":
    asyncio.run(main())

# sheets_manager.py に追加するメソッド
async def get_tasks(self, status_filter="pending"):
    """指定されたステータスのタスクを取得"""
    try:
        worksheet = await self.get_worksheet("Tasks")
        records = worksheet.get_all_records()
        
        filtered_tasks = [
            {**record, 'id': idx+2}  # 行番号をIDとして追加
            for idx, record in enumerate(records)
            if record.get('status', '').lower() == status_filter.lower()
        ]
        
        return filtered_tasks
        
    except Exception as e:
        print(f"❌ タスク取得エラー: {e}")
        return []

async def update_task_result(self, task_id, status, message=""):
    """タスク結果を更新"""
    try:
        worksheet = await self.get_worksheet("Tasks")
        
        # ステータスとメッセージを更新
        worksheet.update_cell(task_id, 6, status)  # ステータス列
        worksheet.update_cell(task_id, 7, message)  # メッセージ列
        worksheet.update_cell(task_id, 8, datetime.now().isoformat())  # 完了日時
        
        print(f"📝 タスク {task_id} を更新: {status}")
        
    except Exception as e:
        print(f"❌ タスク更新エラー: {e}")

import asyncio
import os
import sys
from playwright.async_api import async_playwright
import json

class GeminiBrowserFixed:
    def __init__(self):
        self.browser = None
        self.context = None
        self.page = None
        self.cookie_file = "gemini_cookies.json"
    
    async def initialize(self, headless=False):
        """ブラウザ初期化 - タイムアウト対策版"""
        print("🌐 ブラウザ初期化中...")
        
        # DISPLAY環境変数設定
        os.environ['DISPLAY'] = ':1'
        
        self.playwright = await async_playwright().start()
        
        # ブラウザ起動（ヘッドレスモード調整）
        self.browser = await self.playwright.chromium.launch(
            headless=headless,
            args=[
                '--no-sandbox',
                '--disable-dev-shm-usage',
                '--disable-gpu',
                '--window-size=1150,650',
                '--start-maximized'
            ]
        )
        
        # コンテキスト作成（ビューポート設定）
        self.context = await self.browser.new_context(
            viewport={'width': 1150, 'height': 650},
            user_agent='Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36'
        )
        
        # ページ作成
        self.page = await self.context.new_page()
        
        # ネットワークタイムアウト設定
        self.page.set_default_timeout(60000)  # 60秒に延長
        self.page.set_default_navigation_timeout(60000)
        
        print("✅ ブラウザ初期化完了")
    
    async def load_cookies(self):
        """Cookie読み込み"""
        try:
            if os.path.exists(self.cookie_file):
                with open(self.cookie_file, 'r') as f:
                    cookies = json.load(f)
                    await self.context.add_cookies(cookies)
                print("🍪 Cookie読み込み完了")
                return True
            else:
                print("ℹ️ Cookieファイルなし - 新規セッション")
                return False
        except Exception as e:
            print(f"❌ Cookie読み込みエラー: {e}")
            return False
    
    async def save_cookies(self):
        """Cookie保存"""
        try:
            cookies = await self.context.cookies()
            with open(self.cookie_file, 'w') as f:
                json.dump(cookies, f)
            print("💾 Cookie保存完了")
        except Exception as e:
            print(f"❌ Cookie保存エラー: {e}")
    
    async def navigate_to_gemini(self):
        """Geminiにアクセス - リトライ機能付き"""
        max_retries = 3
        for attempt in range(max_retries):
            try:
                print(f"🔗 Geminiにアクセス中... (試行 {attempt + 1}/{max_retries})")
                
                # ネットワークアイドルを待たずにロード完了を待つ
                await self.page.goto(
                    "https://gemini.google.com/app",
                    wait_until="domcontentloaded",  # networkidle から変更
                    timeout=45000  # 45秒
                )
                
                # 追加の待機（ページ完全読み込み用）
                await self.page.wait_for_timeout(5000)
                
                print("✅ Geminiアクセス成功")
                return True
                
            except Exception as e:
                print(f"❌ アクセス失敗 {attempt + 1}: {e}")
                if attempt < max_retries - 1:
                    print("🔄 リトライ...")
                    await asyncio.sleep(5)
                else:
                    print("💥 最大リトライ回数超過")
                    return False
    
    async def test_gemini_chat(self, message="こんにちは！簡単な自己紹介をしてください"):
        """Geminiチャットテスト"""
        try:
            print(f"💬 チャットテスト: {message}")
            
            # テキストエリアを探して入力
            textarea = await self.page.query_selector('textarea[class*="text-area"]') or \
                      await self.page.query_selector('textarea[placeholder*="メッセージ"]') or \
                      await self.page.query_selector('textarea')
            
            if textarea:
                await textarea.click()
                await textarea.fill(message)
                print("📝 メッセージ入力完了")
                
                # 送信ボタン
                send_button = await self.page.query_selector('button[aria-label*="送信"]') or \
                             await self.page.query_selector('button[class*="send"]') or \
                             await self.page.query_selector('button:has-text("送信")')
                
                if send_button:
                    await send_button.click()
                    print("🚀 メッセージ送信")
                    
                    # 応答待機
                    await self.page.wait_for_timeout(10000)
                    print("✅ 応答受信完了")
                    
                    return True
                else:
                    print("❌ 送信ボタンが見つかりません")
                    return False
            else:
                print("❌ テキストエリアが見つかりません")
                return False
                
        except Exception as e:
            print(f"❌ チャットテストエラー: {e}")
            return False
    
    async def take_screenshot(self, filename="gemini_screenshot.png"):
        """スクリーンショット撮影"""
        try:
            await self.page.screenshot(path=filename)
            print(f"📸 スクリーンショット保存: {filename}")
        except Exception as e:
            print(f"❌ スクリーンショットエラー: {e}")
    
    async def cleanup(self):
        """クリーンアップ"""
        try:
            if self.browser:
                await self.browser.close()
            if self.playwright:
                await self.playwright.stop()
            print("🔚 クリーンアップ完了")
        except Exception as e:
            print(f"❌ クリーンアップエラー: {e}")

import asyncio
import os
import sys
from playwright.async_api import async_playwright
import json

class GeminiBrowserImproved:
    def __init__(self):
        self.browser = None
        self.context = None
        self.page = None
        self.cookie_file = "gemini_cookies.json"
    
    async def initialize(self, headless=False):
        """ブラウザ初期化 - 改良版"""
        print("🌐 ブラウザ初期化中...")
        
        # DISPLAY環境変数設定
        os.environ['DISPLAY'] = ':1'
        
        self.playwright = await async_playwright().start()
        
        # ブラウザ起動（より安定した設定）
        self.browser = await self.playwright.chromium.launch(
            headless=headless,
            args=[
                '--no-sandbox',
                '--disable-dev-shm-usage',
                '--disable-gpu',
                '--disable-software-rasterizer',
                '--window-size=1150,650',
                '--start-maximized',
                '--disable-web-security',
                '--disable-features=VizDisplayCompositor'
            ],
            timeout=60000
        )
        
        # コンテキスト作成
        self.context = await self.browser.new_context(
            viewport={'width': 1150, 'height': 650},
            user_agent='Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
            ignore_https_errors=True
        )
        
        # ページ作成
        self.page = await self.context.new_page()
        
        # タイムアウト設定を延長
        self.page.set_default_timeout(60000)
        self.page.set_default_navigation_timeout(60000)
        
        print("✅ ブラウザ初期化完了")
    
    async def load_cookies(self):
        """Cookie読み込み"""
        try:
            if os.path.exists(self.cookie_file):
                with open(self.cookie_file, 'r') as f:
                    cookies = json.load(f)
                    await self.context.add_cookies(cookies)
                print("🍪 Cookie読み込み完了")
                return True
            else:
                print("ℹ️ Cookieファイルなし - 新規セッション")
                return False
        except Exception as e:
            print(f"❌ Cookie読み込みエラー: {e}")
            return False
    
    async def save_cookies(self):
        """Cookie保存"""
        try:
            cookies = await self.context.cookies()
            with open(self.cookie_file, 'w') as f:
                json.dump(cookies, f)
            print("💾 Cookie保存完了")
        except Exception as e:
            print(f"❌ Cookie保存エラー: {e}")
    
    async def navigate_to_gemini(self):
        """Geminiにアクセス - 複数戦略"""
        max_retries = 3
        for attempt in range(max_retries):
            try:
                print(f"🔗 Geminiにアクセス中... (試行 {attempt + 1}/{max_retries})")
                
                # 戦略1: domcontentloadedでまずページを開く
                await self.page.goto(
                    "https://gemini.google.com/app",
                    wait_until="domcontentloaded",
                    timeout=45000
                )
                
                # 追加待機（JavaScriptの実行を待つ）
                await self.page.wait_for_timeout(8000)
                
                # ページが完全に読み込まれたか確認
                current_url = self.page.url
                print(f"📄 現在のURL: {current_url}")
                
                # リダイレクトやエラーページをチェック
                if "google.com" in current_url and "app" in current_url:
                    print("✅ Geminiアクセス成功")
                    return True
                else:
                    print(f"⚠️ 予期しないURL: {current_url}")
                    continue
                
            except Exception as e:
                print(f"❌ アクセス失敗 {attempt + 1}: {e}")
                if attempt < max_retries - 1:
                    print("🔄 リトライ...")
                    await asyncio.sleep(5)
                else:
                    print("💥 最大リトライ回数超過")
                    return False
        return False
    
    async def find_and_interact_with_chat(self, message="こんにちは！簡単な自己紹介をしてください"):
        """チャットインターフェースを見つけて対話"""
        try:
            print(f"💬 チャット開始: {message}")
            
            # 複数のセレクタを試す
            selectors = [
                'textarea[class*="text"]',
                'textarea[placeholder*="メッセージ"]',
                'textarea[placeholder*="message"]',
                'textarea[placeholder*="Type"]',
                'textarea',
                'div[contenteditable="true"]',
                'div[role="textbox"]',
                'input[type="text"]'
            ]
            
            text_element = None
            for selector in selectors:
                try:
                    element = await self.page.query_selector(selector)
                    if element:
                        text_element = element
                        print(f"✅ 要素発見: {selector}")
                        break
                except:
                    continue
            
            if text_element:
                # 要素をクリックしてフォーカス
                await text_element.click()
                await self.page.wait_for_timeout(1000)
                
                # テキスト入力（要素の種類に応じて）
                tag_name = await text_element.evaluate('el => el.tagName.toLowerCase()')
                if tag_name == 'textarea' or tag_name == 'input':
                    await text_element.fill(message)
                else:  # contenteditable divなど
                    await text_element.evaluate(f'(el) => el.textContent = "{message}"')
                
                print("📝 メッセージ入力完了")
                
                # 送信ボタン検索
                send_selectors = [
                    'button[aria-label*="送信"]',
                    'button[aria-label*="Send"]',
                    'button[class*="send"]',
                    'button:has-text("送信")',
                    'button:has-text("Send")',
                    'button[type="submit"]',
                    'button[data-testid*="send"]'
                ]
                
                send_button = None
                for selector in send_selectors:
                    try:
                        button = await self.page.query_selector(selector)
                        if button:
                            send_button = button
                            print(f"✅ 送信ボタン発見: {selector}")
                            break
                    except:
                        continue
                
                if send_button:
                    await send_button.click()
                    print("🚀 メッセージ送信完了")
                    
                    # 応答待機（動的待機）
                    print("⏳ 応答待機中...")
                    await self.page.wait_for_timeout(15000)
                    
                    # 応答を確認
                    response_selectors = [
                        'div[class*="message"]',
                        'div[class*="response"]',
                        'div[class*="content"]',
                        'div[data-message-type*="assistant"]'
                    ]
                    
                    for selector in response_selectors:
                        responses = await self.page.query_selector_all(selector)
                        if len(responses) > 0:
                            print(f"✅ 応答を検出: {len(responses)}件")
                            break
                    
                    return True
                else:
                    # 送信ボタンが見つからない場合はEnterキーで送信
                    print("⌨️ Enterキーで送信")
                    await self.page.keyboard.press('Enter')
                    await self.page.wait_for_timeout(15000)
                    return True
            else:
                print("❌ 入力要素が見つかりません")
                # スクリーンショットで状況確認
                await self.page.screenshot(path="debug_no_input.png")
                return False
                
        except Exception as e:
            print(f"❌ チャットエラー: {e}")
            await self.page.screenshot(path="debug_chat_error.png")
            return False
    
    async def take_screenshot(self, filename="gemini_screenshot.png"):
        """スクリーンショット撮影"""
        try:
            await self.page.screenshot(path=filename, full_page=True)
            print(f"📸 スクリーンショット保存: {filename}")
        except Exception as e:
            print(f"❌ スクリーンショットエラー: {e}")
    
    async def cleanup(self):
        """クリーンアップ"""
        try:
            if self.browser:
                await self.browser.close()
            if self.playwright:
                await self.playwright.stop()
            print("🔚 クリーンアップ完了")
        except Exception as e:
            print(f"❌ クリーンアップエラー: {e}")

async def main():
    print("=" * 80)
    print("🚀 Geminiブラウザ自動化テスト（完全改良版）")
    print("=" * 80)
    
    gemini = GeminiBrowserImproved()
    
    try:
        # 初期化
        await gemini.initialize(headless=False)
        
        # Cookie読み込み
        await gemini.load_cookies()
        
        # Geminiにアクセス
        success = await gemini.navigate_to_gemini()
        
        if success:
            # アクセス後のスクリーンショット
            await gemini.take_screenshot("gemini_loaded.png")
            
            # チャットテスト
            chat_success = await gemini.find_and_interact_with_chat()
            
            if chat_success:
                # チャット後のスクリーンショット
                await gemini.take_screenshot("gemini_chat_completed.png")
                print("🎉 チャットテスト成功！")
            else:
                print("💥 チャットテスト失敗")
                # デバッグ用スクリーンショット
                await gemini.take_screenshot("gemini_chat_failed.png")
            
            # Cookie保存
            await gemini.save_cookies()
        else:
            print("💥 Geminiアクセス失敗")
            await gemini.take_screenshot("gemini_access_failed.png")
        
    except Exception as e:
        print(f"🏁 main ❌ ERROR ❌ エラー: {e}")
        import traceback
        traceback.print_exc()
    
    finally:
        await gemini.cleanup()
        print("✅ 終了")

if __name__ == "__main__":
    asyncio.run(main())
